cmake_minimum_required(VERSION 3.20)
project(LotGUI_Sample VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(LOTGUI_SDK_ROOT "" CACHE PATH "Installed DesktopSDK prefix, e.g. sdk/windows-x64-msvc")
set(LOTGUI_FONT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/resources/fonts/NotoSansKR-Regular.ttf"
    CACHE FILEPATH "Licensed Noto Sans KR font used by native examples")
option(LOTGUI_BUILD_NATIVE_EXAMPLES "Build native Vulkan window, list and ribbon examples" ON)
if(LOTGUI_SDK_ROOT)
    if(NOT EXISTS "${LOTGUI_SDK_ROOT}/lib/cmake/LotUI/LotUIConfig.cmake")
        message(FATAL_ERROR "LOTGUI_SDK_ROOT must point to a built DesktopSDK, not the headers-only sdk/include directory. See docs/sdk-distribution.md.")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${LOTGUI_SDK_ROOT}")
endif()
# Never add_subdirectory() the private source tree or compile framework .cpp files.
find_package(LotUI 0.1 CONFIG REQUIRED)
include(cmake/SampleHelpers.cmake)
add_executable(custom_draw_contract examples/custom_draw/contract.cpp)
target_link_libraries(custom_draw_contract PRIVATE LotUI::Core)
lotgui_compile_options(custom_draw_contract)
if(BUILD_TESTING)
    add_test(NAME custom_draw_contract COMMAND custom_draw_contract)
endif()
if(LOTGUI_BUILD_NATIVE_EXAMPLES)
    foreach(_target LotUI::Widgets LotUI::Platform LotUI::Renderer LotUI::Vulkan LotUI::TextFreeType)
        if(NOT TARGET ${_target})
            message(FATAL_ERROR "${_target} missing: export a full DesktopSDK with Vulkan and FreeType enabled, or set LOTGUI_BUILD_NATIVE_EXAMPLES=OFF for the headless custom-draw contract.")
        endif()
    endforeach()
    if(NOT EXISTS "${LOTGUI_FONT_FILE}")
        message(FATAL_ERROR "Set LOTGUI_FONT_FILE to an official Noto Sans KR font. It is not silently downloaded or substituted.")
    endif()
    set(_bridge examples/platform_probe/platform_event_bridge.cpp)
    add_executable(platform_probe examples/platform_probe/widget_main.cpp
        examples/platform_probe/demo_ui.cpp ${_bridge})
    target_compile_definitions(platform_probe PRIVATE LOTUI_HAS_EXAMPLE_TEXT=1)
    add_executable(list_control_demo examples/list_control_demo/main.cpp ${_bridge})
    if(WIN32)
        set(_loader examples/ribbon_demo/plugin_loader_win32.cpp)
    else()
        set(_loader examples/ribbon_demo/plugin_loader_posix.cpp)
    endif()
    add_library(ribbon_demo_plugin MODULE examples/ribbon_demo/plugin_module.cpp)
    target_include_directories(ribbon_demo_plugin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lotgui_compile_options(ribbon_demo_plugin)
    add_executable(ribbon_demo examples/ribbon_demo/main.cpp
        examples/ribbon_demo/plugin_tab.cpp ${_loader} ${_bridge})
    target_link_libraries(ribbon_demo PRIVATE ${CMAKE_DL_LIBS})
    target_compile_definitions(ribbon_demo PRIVATE
        LOTUI_DEMO_PLUGIN_NAME="$<TARGET_FILE_NAME:ribbon_demo_plugin>")
    add_dependencies(ribbon_demo ribbon_demo_plugin)
    add_executable(custom_draw_demo examples/custom_draw/main.cpp ${_bridge})
    foreach(_target platform_probe list_control_demo ribbon_demo custom_draw_demo)
        target_include_directories(${_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(${_target} PRIVATE LotUI::Vulkan LotUI::Widgets LotUI::TextFreeType)
        lotgui_compile_options(${_target})
        lotgui_stage_resources(${_target})
        install(TARGETS ${_target} RUNTIME DESTINATION bin)
    endforeach()
    add_custom_command(TARGET ribbon_demo POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:ribbon_demo_plugin>" "$<TARGET_FILE_DIR:ribbon_demo>"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/examples/ribbon_demo/resources/fonts/lucide-subset.ttf"
            "$<TARGET_FILE_DIR:ribbon_demo>/resources/fonts/lucide-subset.ttf"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/examples/ribbon_demo/resources/LUCIDE-LICENSE.txt"
            "$<TARGET_FILE_DIR:ribbon_demo>/legal/licenses/LUCIDE-LICENSE.txt"
        VERBATIM)
    install(TARGETS ribbon_demo_plugin LIBRARY DESTINATION bin RUNTIME DESTINATION bin)
    install(FILES "${LOTGUI_FONT_FILE}" DESTINATION bin/resources/fonts RENAME NotoSansKR-Regular.ttf)
    install(FILES examples/ribbon_demo/resources/fonts/lucide-subset.ttf DESTINATION bin/resources/fonts)
    install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION bin/legal)
    install(DIRECTORY licenses/ DESTINATION bin/legal/licenses)
    install(FILES examples/ribbon_demo/resources/LUCIDE-LICENSE.txt DESTINATION bin/legal/licenses)
    if(BUILD_TESTING)
        add_test(NAME list_control_demo_state_tests COMMAND list_control_demo --self-test)
    endif()
endif()
